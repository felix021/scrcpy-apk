cmake_minimum_required(VERSION 3.22)
project(scrcpy_adb_crypto LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
add_compile_options("-ffile-prefix-map=${REPO}=.")
# Compile BoringSSL separately: AGP 7 cannot model its multi-object FIPS target.
foreach(component ssl crypto)
    add_library(${component} STATIC IMPORTED)
    set_target_properties(${component} PROPERTIES IMPORTED_LOCATION "${REPO}/build/boringssl/${ANDROID_ABI}/lib${component}.a")
endforeach()
add_library(adbcrypto SHARED AdbCrypto.c jni.c)
target_include_directories(adbcrypto PRIVATE include "${REPO}/build/boringssl/boringssl-0131ae9d78e42fcc2c1811315e62410ad96d8230/include")
target_link_libraries(adbcrypto PRIVATE ssl crypto)
target_link_options(adbcrypto PRIVATE "-Wl,--exclude-libs,ALL" "-Wl,-Bsymbolic")
set_target_properties(adbcrypto PROPERTIES LINKER_LANGUAGE CXX C_VISIBILITY_PRESET hidden CXX_VISIBILITY_PRESET hidden)
